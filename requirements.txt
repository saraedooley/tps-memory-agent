# Databricks Apps runtime dependencies (CPython 3.11).
#
# The app imports, all from `databricks_langchain`: ChatDatabricks,
# DatabricksEmbeddings, UCFunctionToolkit, AsyncDatabricksStore,
# AsyncCheckpointSaver (the `[memory]` extra is REQUIRED for the async store +
# checkpointer) -- plus langgraph, langchain-core, mlflow tracing, fastapi/uvicorn.
#
# Why these exact pins: an unpinned LangChain stack sent the Apps pip build into
# `resolution-too-deep` -- pip explored 60+ `langsmith` candidates (dragged in by
# `langchain-community`) and 40+ `langchain-core` versions before timing out after
# ~47 min. The pins below come from databricks/databricks-ai-bridge's own
# `uv.lock` at the `databricks-langchain-v0.17.0` tag, a known-consistent
# resolution, so pip does ZERO backtracking on the LangChain stack.
#
# Only PURE-PYTHON packages are pinned (version-agnostic across CPython builds).
# Native / py-version-specific deps (databricks-connect, grpcio, pyarrow,
# pydantic-core, psycopg-binary, numpy, sqlalchemy, starlette, ...) and tightly
# version-coupled ones (databricks-sdk <-> databricks-connect, pydantic) are left
# unpinned on purpose so pip selects cp311 wheels and mutually compatible versions.

databricks-ai-bridge[memory]==0.17.0
unitycatalog-langchain[databricks]==0.3.0
